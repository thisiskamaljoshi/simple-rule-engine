from rule import Rule
from rule_engine import evaluate_logical_rule, fire_rule

# 1. Declarative Business / Medical Rules
rules = [
    Rule(
        conditions=[
            "fever==True AND temperature>=38",
            "body_weakness==True AND spots==False"
        ],
        conclusion="Viral Fever"
    )
]

# 2. Dynamic Working Memory (Facts)
patient_facts = {
    "fever": True,
    "temperature": 39,
    "body_weakness": True,
    "spots": False
}

# 3. Safe, zero-eval condition parsing & rule firing
for rule in rules:
    matched = evaluate_logical_rule(rule, patient_facts)
    fire_rule(rule, matched)
# Output: -> "Viral Fever"
